# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format I: Register-Register (16-bit)
# Encoding: bits[15-11]=reg2, bits[10-5]=opcode, bits[4-0]=reg1
#
# Note: Each 16-bit instruction is paired with NOP (0x00,0x00) to form 4 bytes,
# preventing incorrect 32-bit instruction decoding.
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# NOP - encoded as MOV r0, r0
#------------------------------------------------------------------------------

# CHECK: nop
0x00,0x00

#------------------------------------------------------------------------------

# CHECK: mov r0, r1
# CHECK: nop

# CHECK: mov r1, r2
# CHECK: nop

# CHECK: mov r31, r30
# CHECK: nop
