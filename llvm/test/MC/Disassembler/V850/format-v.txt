# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format V: Jump (32-bit)
# Encoding: First halfword: bits[15-11]=reg2, bits[10-6]=opcode[5:1], bits[5-0]=disp22[5:0]
#           Second halfword: bits[31-17]=disp22[21:7], bits[16]=disp22[6]
#
# Note: disp22 bit 0 is implicitly 0 (halfword aligned)
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# JR disp22 - Jump relative (unconditional)
# reg2 field = 0
#------------------------------------------------------------------------------

# CHECK: jr	100
0xf2,0x05,0x00,0x00

# CHECK: jr	1000
0xf4,0x05,0x07,0x00

#------------------------------------------------------------------------------
# JARL disp22, reg2 - Jump and register link
#------------------------------------------------------------------------------

# CHECK: jarl	100, r10
0xf2,0x55,0x00,0x00

# CHECK: jarl	2000, r20
0xe8,0xa5,0x0f,0x00
