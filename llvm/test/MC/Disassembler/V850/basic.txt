# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format I: Register-Register (16-bit)
# Note: Each 16-bit instruction is followed by a 16-bit NOP to prevent
# incorrect 32-bit decoding of consecutive 16-bit instruction bytes.
#------------------------------------------------------------------------------

# CHECK: add r3, r5

# CHECK: sub r4, r6

# CHECK: and r7, r8

# CHECK: or r9, r10

0x09,0x51,0x00,0x00

# CHECK: xor r11, r12

# CHECK: not r13, r14

# CHECK: jmp [r20]
