# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format VI: 3-operand with 16-bit immediate (32-bit)
# Encoding: First halfword: bits[15-11]=reg2, bits[10-5]=opcode, bits[4-0]=reg1
#           Second halfword: bits[31-16]=imm16
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# ADDI imm16, reg1, reg2 - Add immediate (opcode: 0x30)
# imm16 is sign-extended
#------------------------------------------------------------------------------

# CHECK: addi	100, r1, r2
0x01,0x16,0x64,0x00

# CHECK: addi	-100, r5, r10
0x05,0x56,0x9c,0xff

#------------------------------------------------------------------------------
# MOVEA imm16, reg1, reg2 - Move effective address (opcode: 0x31)
# imm16 is sign-extended
#------------------------------------------------------------------------------

# CHECK: movea	200, r3, r4
0x23,0x26,0xc8,0x00

#------------------------------------------------------------------------------
# MOVHI imm16, reg1, reg2 - Move high (opcode: 0x32)
# imm16 shifted left by 16 and added to reg1
#------------------------------------------------------------------------------

# CHECK: movhi	4660, r5, r6
0x45,0x36,0x34,0x12

#------------------------------------------------------------------------------
# ORI imm16, reg1, reg2 - OR immediate (opcode: 0x34)
# imm16 is zero-extended
#------------------------------------------------------------------------------

# CHECK: ori	255, r7, r8
0x87,0x46,0xff,0x00

#------------------------------------------------------------------------------
# XORI imm16, reg1, reg2 - XOR immediate (opcode: 0x35)
# imm16 is zero-extended
#------------------------------------------------------------------------------

# CHECK: xori	170, r9, r10
0xa9,0x56,0xaa,0x00

#------------------------------------------------------------------------------
# ANDI imm16, reg1, reg2 - AND immediate (opcode: 0x36)
# imm16 is zero-extended
#------------------------------------------------------------------------------

# CHECK: andi	85, r11, r12
0xcb,0x66,0x55,0x00

#------------------------------------------------------------------------------
# MULHI imm16, reg1, reg2 - Multiply high immediate (opcode: 0x37)
#------------------------------------------------------------------------------

# CHECK: mulhi	100, r5, r10
0xe5,0x56,0x64,0x00
