# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format IX: Extended 1 (32-bit)
# Encoding: First halfword: bits[15-11]=reg2, bits[10-5]=opcode(0x3F), bits[4-0]=reg1
#           Second halfword: bits[31-27]=0, bits[26-21]=sub-op, bits[20-16]=0
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# SHR reg1, reg2 - Shift right logical by register (sub-op: 0x02)
#------------------------------------------------------------------------------

# CHECK: shr	r1, r2
0xe1,0x17,0x40,0x00

# CHECK: shr	r5, r10
0xe5,0x57,0x40,0x00

#------------------------------------------------------------------------------
# SAR reg1, reg2 - Shift right arithmetic by register (sub-op: 0x04)
#------------------------------------------------------------------------------

# CHECK: sar	r3, r4
0xe3,0x27,0x80,0x00

# CHECK: sar	r10, r20
0xea,0xa7,0x80,0x00

#------------------------------------------------------------------------------
# SHL reg1, reg2 - Shift left logical by register (sub-op: 0x06)
#------------------------------------------------------------------------------

# CHECK: shl	r5, r6
0xe5,0x37,0xc0,0x00

# CHECK: shl	r20, r25
0xf4,0xcf,0xc0,0x00

#------------------------------------------------------------------------------
# LDSR reg2, regID - Load to system register (sub-op: 0x20)
#------------------------------------------------------------------------------

# CHECK: ldsr	r5, psw
0xe5,0x2f,0x00,0x04

#------------------------------------------------------------------------------
# STSR regID, reg2 - Store from system register (sub-op: 0x24)
#------------------------------------------------------------------------------

# CHECK: stsr	psw, r10
0xe5,0x57,0x80,0x04
