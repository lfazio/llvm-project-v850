# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format VII: Load/Store (32-bit)
# Encoding: First halfword: bits[15-11]=reg2, bits[10-5]=opcode, bits[4-0]=reg1
#           Second halfword: bits[31-17]=disp16[15:1], bits[16]=disp16[0] or sub-op
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# LD.B disp16[reg1], reg2 - Load byte sign-extended (opcode: 0x38)
#------------------------------------------------------------------------------

# CHECK: ld.b	0[r1], r2
0x01,0x17,0x00,0x00

# CHECK: ld.b	100[r5], r10
0x05,0x57,0x64,0x00

#------------------------------------------------------------------------------
# LD.H disp16[reg1], reg2 - Load halfword sign-extended (opcode: 0x39, bit16=0)
#------------------------------------------------------------------------------

# CHECK: ld.h	4[r3], r4
0x23,0x27,0x04,0x00

#------------------------------------------------------------------------------
# LD.W disp16[reg1], reg2 - Load word (opcode: 0x39, bit16=1)
#------------------------------------------------------------------------------

# CHECK: ld.w	8[r5], r6
0x25,0x37,0x09,0x00

#------------------------------------------------------------------------------
# ST.B reg2, disp16[reg1] - Store byte (opcode: 0x3A)
#------------------------------------------------------------------------------

# CHECK: st.b	r7, 0[r8]
0x48,0x3f,0x00,0x00

#------------------------------------------------------------------------------
# ST.H reg2, disp16[reg1] - Store halfword (opcode: 0x3B, bit16=0)
#------------------------------------------------------------------------------

# CHECK: st.h	r9, 4[r10]
0x6a,0x4f,0x04,0x00

#------------------------------------------------------------------------------
# ST.W reg2, disp16[reg1] - Store word (opcode: 0x3B, bit16=1)
#------------------------------------------------------------------------------

# CHECK: st.w	r11, 8[r12]
0x6c,0x5f,0x09,0x00
