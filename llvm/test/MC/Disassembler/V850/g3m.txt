# RUN: llvm-mc -triple=v850-unknown-elf -mcpu=g3m --disassemble %s | FileCheck %s

# Test RH850G3M specific instructions disassembly

# SYNCI - Synchronize instruction pipeline
# CHECK: synci
0x1c 0x00

# SNOOZE - Enter low-power snooze state
# CHECK: snooze
0xe0 0x0f 0x20 0x01

# CLL - Clear load link state
# CHECK: cll
0xff 0xff 0x60 0xf1

# LDL.W - Load linked word
# CHECK: ldl.w [r10], r11
0xea 0x07 0x78 0x5b

# CHECK: ldl.w [r0], r1
0xe0 0x07 0x78 0x0b

# CHECK: ldl.w [r31], r31
0xff 0x07 0x78 0xfb

# STC.W - Store conditional word
# CHECK: stc.w r11, [r10]
0xea 0x07 0x7a 0x5b

# CHECK: stc.w r1, [r0]
0xe0 0x07 0x7a 0x0b

# CHECK: stc.w r31, [r31]
0xff 0x07 0x7a 0xfb

# ROTL - Rotate left with immediate
# CHECK: rotl 5, r6, r10
0xe5 0x37 0xc4 0x50

# CHECK: rotl 0, r0, r0
0xe0 0x07 0xc4 0x00

# CHECK: rotl 31, r31, r31
0xff 0xff 0xc4 0xf8

# ROTL - Rotate left with register
# CHECK: rotl r7, r6, r10
0xe7 0x37 0xc6 0x50

# CHECK: rotl r0, r0, r0
0xe0 0x07 0xc6 0x00

# CHECK: rotl r31, r31, r31
0xff 0xff 0xc6 0xf8

# Bcond disp17 - Extended conditional branches (32-bit encoding)
# Format: 00000111111DCCCC ddddddddddddddd1

# BV disp17 - Overflow

0xe0 0x07 0x01 0x01

# CHECK: bv -258
0xf0 0x07 0xff 0xfe

# CHECK: bc 500
0xe1 0x07 0xf5 0x01

# BZ disp17 - Zero
# CHECK: bz 1000
0xe2 0x07 0xe9 0x03

# BNH disp17 - Not higher
# CHECK: bnh 2000
0xe3 0x07 0xd1 0x07

# BN disp17 - Negative
# CHECK: bn -2000
0xf4 0x07 0x31 0xf8

# BLT disp17 - Less than (signed)
# CHECK: blt 4000
0xe6 0x07 0xa1 0x0f

# BLE disp17 - Less or equal (signed)
# CHECK: ble -4000
0xf7 0x07 0x61 0xf0

# BNV disp17 - No overflow
# CHECK: bnv 8000
0xe8 0x07 0x41 0x1f

# BNC disp17 - No carry
# CHECK: bnc -8000
0xf9 0x07 0xc1 0xe0

# BNZ disp17 - Not zero
# CHECK: bnz 16000
0xea 0x07 0x81 0x3e

# BH disp17 - Higher
# CHECK: bh -16000
0xfb 0x07 0x81 0xc1

# BP disp17 - Positive
# CHECK: bp 32000
0xec 0x07 0x01 0x7d

# BSA disp17 - Saturated

0xee 0x07 0xff 0xff

# BGT disp17 - Greater than (signed)

# PUSHSP reg_list - Push multiple registers onto stack

# POPSP reg_list - Pop multiple registers from stack
# CHECK: popsp r20, r31
0xf4 0x67 0x60 0xf9

# LOOP reg1, disp16 - Decrement and branch if not zero (32-bit)
# Note: bit[16]=1 distinguishes LOOP from JMP32 (bit[16]=0)

# CACHE cacheop, [reg1] - Cache operation

# PREF prefop, [reg1] - Prefetch hint

# Negative displacement
