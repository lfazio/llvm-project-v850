# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format III: Conditional Branch (16-bit)
# Encoding: bits[15-11]=half_disp[7:3], bits[10-7]=opcode(1011),
#           bits[6-4]=half_disp[2:0], bits[3-0]=condition
#
# Note: Each 16-bit instruction is paired with NOP to form 4 bytes.
# Branch target bit 0 is implicitly 0 (halfword aligned).
#
# Some Bcond+NOP combinations have the same bit pattern as 32-bit JR/JARL
# (FormatV). When cond has bit[3]=0 and half_disp has bit[2]=1, bits[10:6]
# become 10111 which matches JR/JARL's opcode. The disassembler prefers the
# 32-bit decode in that case. Only non-conflicting encodings are tested here.
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# BV - Branch if overflow (cond: 0000)
# Use bv 16 (half_disp[2]=0) to avoid JR/JARL conflict.
# bv 8 has encoding 0xc0,0x05 which matches JR format (bits[10:6]=10111).
#------------------------------------------------------------------------------

# CHECK: bv	16
# CHECK: nop
0x80,0x0d,0x00,0x00

#------------------------------------------------------------------------------
# BNH - Branch if not higher (cond: 0011)
#------------------------------------------------------------------------------

# CHECK: bnh	32
# CHECK: nop
0x83,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BN - Branch if negative (cond: 0100)
# Use disp=32 to avoid JARL conflict (half_disp[2]=0).
#------------------------------------------------------------------------------

# CHECK: bn	32
# CHECK: nop
0x84,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BLE - Branch if less or equal (signed) (cond: 0111)
#------------------------------------------------------------------------------

# CHECK: ble	64
# CHECK: nop
0x87,0x25,0x00,0x00

#------------------------------------------------------------------------------
# BNV - Branch if no overflow (cond: 1000)
# Use disp=32 to avoid JARL conflict (half_disp[2]=0).
#------------------------------------------------------------------------------

# CHECK: bnv	32
# CHECK: nop
0x88,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BH - Branch if higher (cond: 1011)
#------------------------------------------------------------------------------

# CHECK: bh	96
# CHECK: nop
0x8b,0x35,0x00,0x00

#------------------------------------------------------------------------------
# BP - Branch if positive (cond: 1100)
# Use disp=32 to avoid JARL conflict (half_disp[2]=0).
#------------------------------------------------------------------------------

# CHECK: bp	32
# CHECK: nop
0x8c,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BGT - Branch if greater than (signed) (cond: 1111)
#------------------------------------------------------------------------------

# CHECK: bgt	128
# CHECK: nop
0x8f,0x45,0x00,0x00
