# RUN: llvm-mc -disassemble %s -triple=v850 | FileCheck %s

#------------------------------------------------------------------------------
# Format III: Conditional Branch (16-bit)
# Encoding: bits[15-11]=half_disp[7:3], bits[10-7]=opcode(1011),
#           bits[6-4]=half_disp[2:0], bits[3-0]=condition
#
# Note: Each 16-bit instruction is paired with NOP to form 4 bytes.
# Branch target bit 0 is implicitly 0 (halfword aligned).
#
# Some branch+NOP combinations conflict with 32-bit JR/JARL instruction patterns.
# Only non-conflicting branches are tested here to avoid decoder ambiguity.
#------------------------------------------------------------------------------

#------------------------------------------------------------------------------
# BV - Branch if overflow (cond: 0000)
#------------------------------------------------------------------------------

# CHECK: bv	8
# CHECK: nop
0xc0,0x05,0x00,0x00

#------------------------------------------------------------------------------
# BNH - Branch if not higher (cond: 0011)
#------------------------------------------------------------------------------

# CHECK: bnh	32
# CHECK: nop
0x83,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BN - Branch if negative (cond: 0100)
#------------------------------------------------------------------------------

# CHECK: bn	40
# CHECK: nop
0xc4,0x15,0x00,0x00

#------------------------------------------------------------------------------
# BLE - Branch if less or equal (signed) (cond: 0111)
#------------------------------------------------------------------------------

# CHECK: ble	64
# CHECK: nop
0x87,0x25,0x00,0x00

#------------------------------------------------------------------------------
# BNV - Branch if no overflow (cond: 1000)
#------------------------------------------------------------------------------

# CHECK: bnv	72
# CHECK: nop
0xc8,0x25,0x00,0x00

#------------------------------------------------------------------------------
# BH - Branch if higher (cond: 1011)
#------------------------------------------------------------------------------

# CHECK: bh	96
# CHECK: nop
0x8b,0x35,0x00,0x00

#------------------------------------------------------------------------------
# BP - Branch if positive (cond: 1100)
#------------------------------------------------------------------------------

# CHECK: bp	104
# CHECK: nop
0xcc,0x35,0x00,0x00

#------------------------------------------------------------------------------
# BGT - Branch if greater than (signed) (cond: 1111)
#------------------------------------------------------------------------------

# CHECK: bgt	128
# CHECK: nop
0x8f,0x45,0x00,0x00
